// chat-ai.js var step = 0; var userName = ''; var intention = ''; var persona = ''; var mode = ''; var pendingIdea = ''; var pendingBlocker = ''; var chatHistory = document.getElementById('chat-history'); var userInput = document.getElementById('user-input'); var sendBtn = document.getElementById('send-btn'); var activeIntent = document.getElementById('active-intent'); var activePersona = document.getElementById('active-persona'); var ideaBoxContent = document.getElementById('idea-box-content'); var blockerBoxContent = document.getElementById('blocker-box-content'); var outputContent = document.getElementById('output-content'); var energyBar = document.getElementById('energy-bar'); var energyText = document.getElementById('energy-text'); function safeElementCheck() { // 如果任何必要元素唔存在，記錄並停止初始化互動（避免 runtime error） var required = [ { el: chatHistory, name: 'chat-history' }, { el: userInput, name: 'user-input' }, { el: sendBtn, name: 'send-btn' }, { el: outputContent, name: 'output-content' } ]; for (var i = 0; i < required.length; i++) { if (!required[i].el) { console.warn('Missing DOM element:', required[i].name); return false; } } return true; } function appendAiMessage(text) { if (!chatHistory) return; var div = document.createElement('div'); div.className = 'flex items-start space-x-3 max-w-[85%] animate-fade-in'; var avatar = document.createElement('div'); avatar.className = 'bg-blue-600/20 text-blue-400 p-2 rounded-xl font-bold text-xs border border-blue-500/20'; avatar.innerText = 'AI'; var contentBox = document.createElement('div'); contentBox.className = 'bg-slate-900 border border-slate-800 rounded-2xl rounded-tl-none p-4 shadow-md text-slate-200 text-sm leading-relaxed whitespace-pre-wrap'; contentBox.innerText = text; div.appendChild(avatar); div.appendChild(contentBox); chatHistory.appendChild(div); chatHistory.scrollTop = chatHistory.scrollHeight; } function appendUserMessage(text) { if (!chatHistory) return; var div = document.createElement('div'); div.className = 'flex items-start space-x-3 max-w-[85%] ml-auto justify-end animate-fade-in'; var contentBox = document.createElement('div'); contentBox.className = 'bg-slate-800 border border-slate-700 rounded-2xl rounded-tr-none p-4 shadow-md text-slate-200 text-sm leading-relaxed'; contentBox.innerText = text; var avatar = document.createElement('div'); avatar.className = 'bg-slate-700 text-slate-200 p-2 rounded-xl font-bold text-xs border border-slate-600'; avatar.innerText = 'ME'; div.appendChild(contentBox); div.appendChild(avatar); chatHistory.appendChild(div); chatHistory.scrollTop = chatHistory.scrollHeight; } function handleDialogue(input) { if (!input) return; if (step === 0) { userName = input.trim() || 'User'; step = 1; appendAiMessage('Nice to meet you, ' + userName + '!\n\nLet\'s cut through the noise. What is your Intention today? What business pain point or product idea are we trying to refine? (e.g., "I want to automate our VIP retail customer routing")'); return; } if (step === 1) { intention = input.trim(); if (activeIntent) { activeIntent.innerText = intention; activeIntent.classList.remove('italic', 'text-slate-400'); activeIntent.classList.add('text-blue-400', 'font-medium'); } step = 2; appendAiMessage('Got it. Intent locked as: "' + intention + '"\n\nNext, what persona should I adopt to test this idea? Choose your opponent:\n\n1. [The Tough Boss] (Will grill you relentlessly on ROI and business value)\n2. [The Realistic Engineer] (Will challenge your data infrastructure and scalability)\n3. [The Frontline Manager] (Will test how this impacts real daily operations)'); return; } if (step === 2) { var choices = { '1': 'Tough Boss', '2': 'Realistic Engineer', '3': 'Frontline Manager' }; persona = choices[input.trim()] || input.trim(); if (activePersona) { activePersona.innerText = persona; activePersona.classList.remove('italic', 'text-slate-400'); activePersona.classList.add('text-indigo-400', 'font-medium'); } step = 3; appendAiMessage('Understood. I am now acting as your [' + persona + '].\n\nFinal setting: What Thinking Mode do you require?\n\n1. [Concise Mode] (Short answers, high efficiency, lets you speak more)\n2. [Deep Dialectic Mode] (Deep critical analysis, complex questioning, treats you like an executive)'); return; } if (step === 3) { mode = input.trim() === '1' ? 'Concise' : 'Deep Dialectic'; step = 4; if (persona.includes('Engineer')) { appendAiMessage('⚡ [System Notification: Session Initialized in ' + mode + ' Mode]\n\nAlright ' + userName + ', let\'s talk architecture. You want to execute: "' + intention + '". \n\nMy first question to you: Do we actually have clean, structured data for this? If we feed dirty CRM logs or raw PDFs into an LLM right now, it will throw a wall of hallucinations. Do you have an established data pipeline, or are we building on a mess?'); } else if (persona.includes('Boss')) { appendAiMessage('⚡ [System Notification: Session Initialized in ' + mode + ' Mode]\n\nOkay ' + userName + ', let\'s look at the numbers. To build "' + intention + '", we are going to burn cash on API tokens and engineering hours. \n\nTell me: What is the exact ROI here? Are we trying to cut headcounts, speed up conversion rates, or is this just a vanity project to show off to the board? Be brutally honest.'); } else { appendAiMessage('⚡ [System Notification: Session Initialized in ' + mode + ' Mode]\n\nLet\'s get real, ' + userName + '. Frontline staff hate new software. For "' + intention + '", if this AI takes an extra 3 steps to use, the team will sabotage it or bypass it completely. \n\nTell me: How exactly does a retail rep use this in their busiest hour? What is the exact workflow friction we are solving?'); } setTimeout(function() { triggerIdeaSuggestion('Establish hardcoded filter criteria at API gateway before routing to core model.'); }, 1500); return; } if (step >= 4) { var lowInput = input.toLowerCase(); if ((lowInput.includes('yes') || lowInput.includes('store') || lowInput.includes('save')) && pendingIdea) { commitPendingIdea(); return; } if ((lowInput.includes('yes') || lowInput.includes('capture') || lowInput.includes('anchor')) && pendingBlocker) { commitPendingBlocker(); return; } if (step === 4) { step = 5; updateEnergy(70, '70% Left'); appendAiMessage('That\'s a fair point, but it feels incomplete. If we don\'t lock this into a rigid Finite State Machine (FSM) external code layer, the AI prompt will drift during edge cases.\n\nShould we restrict the tool access so it only outputs pure JSON, or do you want to give the model full conversational freedom?'); setTimeout(function() { triggerBlockerSuggestion('CRITICAL BLOCKER: Core CRM architecture lacks synchronized webhooks. Real-time routing currently impossible.'); }, 1200); } else if (step === 5) { step = 6; updateEnergy(50, '50% Left'); appendAiMessage('Excellent choice. Locking it down via Hard Code protects the brand image. \n\nI think we have enough strategic weight now. Type "Final" or "Generate" anytime, and I will compile our architectural logic flow and defense brief inside the Final Output Window.'); } else if (lowInput.includes('final') || lowInput.includes('generate')) { compileFinalOutput(); } else { appendAiMessage('Understood. Let\'s dig deeper into that logic. How do you plan to handle liability if the system misinterprets a tier-1 customer\'s intent?'); } } } function triggerIdeaSuggestion(ideaText) { pendingIdea = ideaText; if (!chatHistory) return; var div = document.createElement('div'); div.className = 'bg-blue-950/40 border border-blue-800/60 rounded-xl p-3 mt-2 text-xs space-y-2 animate-fade-in'; var title = document.createElement('p'); title.className = 'text-blue-300 font-medium'; title.innerHTML = ' [AI Insight Extraction] Would you like to store this concept into your Idea Storage Box?'; var textP = document.createElement('p'); textP.className = 'text-slate-300 italic'; textP.innerText = '"' + ideaText + '"'; var btnGroup = document.createElement('div'); btnGroup.className = 'flex space-x-2 pt-1'; // 使用安全的 onclick handler var yesBtn = document.createElement('button'); yesBtn.className = 'bg-blue-600 text-white px-3 py-1 rounded font-medium'; yesBtn.innerText = 'Yes, Store It'; yesBtn.addEventListener('click', commitPendingIdea); var noBtn = document.createElement('button'); noBtn.className = 'text-slate-500 px-2 py-1'; noBtn.innerText = 'Discard'; noBtn.addEventListener('click', rejectPendingIdea); btnGroup.appendChild(yesBtn); btnGroup.appendChild(noBtn); div.appendChild(title); div.appendChild(textP); div.appendChild(btnGroup); chatHistory.appendChild(div); chatHistory.scrollTop = chatHistory.scrollHeight; } function commitPendingIdea() { if (!pendingIdea) return; if (!ideaBoxContent) return; if (ideaBoxContent.innerText.includes('No approved ideas')) { ideaBoxContent.innerHTML = ''; } var item = document.createElement('div'); item.className = 'bg-slate-950 border border-slate-800 rounded-xl p-3 flex justify-between items-start text-xs text-slate-200 font-sans shadow-inner animate-scale-up'; var p = document.createElement('p'); p.className = 'leading-relaxed'; p.innerHTML = ''; var textSpan = document.createElement('span'); textSpan.innerText = pendingIdea; p.appendChild(textSpan); var btn = document.createElement('button'); btn.className = 'text-slate-600 hover:text-blue-400 ml-2'; btn.innerHTML = ''; // 可加入分享或複製功能，暫時只做視覺按鈕 item.appendChild(p); item.appendChild(btn); ideaBoxContent.appendChild(item); pendingIdea = ''; appendAiMessage('🔒 Concept successfully locked into your Idea Storage Box.'); } function rejectPendingIdea() { pendingIdea = ''; appendAiMessage('Discarded. Let\'s keep refining.'); } function triggerBlockerSuggestion(blockerText) { pendingBlocker = blockerText; if (!chatHistory) return; var div = document.createElement('div'); div.className = 'bg-amber-950/40 border border-amber-800/60 rounded-xl p-3 mt-2 text-xs space-y-2 animate-fade-in'; var title = document.createElement('p'); title.className = 'text-amber-400 font-medium'; title.innerHTML = ' [Friction Detected] You seem to be hitting a wall here. Shall we anchor this Blocker into your Question Box?'; var textP = document.createElement('p'); textP.className = 'text-slate-300 italic'; textP.innerText = '"' + blockerText + '"'; var btnGroup = document.createElement('div'); btnGroup.className = 'flex space-x-2 pt-1'; var yesBtn = document.createElement('button'); yesBtn.className = 'bg-amber-600 text-white px-3 py-1 rounded font-medium'; yesBtn.innerText = 'Yes, Capture Blocker'; yesBtn.addEventListener('click', commitPendingBlocker); var ignoreBtn = document.createElement('button'); ignoreBtn.className = 'text-slate-500 px-2 py-1'; ignoreBtn.innerText = 'Ignore'; ignoreBtn.addEventListener('click', rejectPendingBlocker); btnGroup.appendChild(yesBtn); btnGroup.appendChild(ignoreBtn); div.appendChild(title); div.appendChild(textP); div.appendChild(btnGroup); chatHistory.appendChild(div); chatHistory.scrollTop = chatHistory.scrollHeight; } function commitPendingBlocker() { if (!pendingBlocker) return; if (!blockerBoxContent) return; if (blockerBoxContent.innerText.includes('No captured blockers')) { blockerBoxContent.innerHTML = ''; } var item = document.createElement('div'); item.className = 'bg-slate-950 border border-slate-800 rounded-xl p-3 flex justify-between items-start text-xs text-slate-200 font-sans shadow-inner animate-scale-up'; var p = document.createElement('p'); p.className = 'leading-relaxed'; p.innerHTML = ''; var textSpan = document.createElement('span'); textSpan.innerText = pendingBlocker; p.appendChild(textSpan); var btn = document.createElement('button'); btn.className = 'text-slate-600 hover:text-amber-400 ml-2'; btn.innerHTML = ''; item.appendChild(p); item.appendChild(btn); blockerBoxContent.appendChild(item); pendingBlocker = ''; appendAiMessage('⚠️ Friction node safely anchored inside your Question & Blocker Box.'); } function rejectPendingBlocker() { pendingBlocker = ''; appendAiMessage('Understood. Overriding friction node, continuing deep dive.'); } function compileFinalOutput() { if (!outputContent) return; var lines = []; lines.push('// PRE-MEETING STRATEGIC BRIEF GENERATED'); lines.push('[PROJECT INTENT] ' + intention); lines.push('[AI PERSONA REVIEWER] ' + persona + ' (' + mode + ' Mode)'); lines.push(''); lines.push('■ EXECUTIVE PRESENTATION BRIEF (SLIDE SCRIPTS)'); lines.push('Slide 1: The Core Target'); lines.push('Deploy structured AI filtering to streamline "' + intention + '" without expanding human headcount costs.'); lines.push('Slide 2: Architectural Defense Line'); lines.push('Implement strict API Token gateway guardrails. Force JSON output structures to block model hallucination and protect Premier Retail brand image.'); lines.push('Slide 3: Captured Operational Blocker'); lines.push('Acknowledge critical sync friction: Current API pipelines lack synchronized webhooks. Real-time CRM execution requires prior infrastructure correction.'); lines.push(''); lines.push('■ PROPOSED ACTION LOGIC FLOW'); lines.push('User Input ➔ Gateway Hard Filter ➔ Intent Router (JSON Only) ➔ Human-in-the-Loop Backup Verification.'); outputContent.innerText = lines.join('\n\n'); appendAiMessage('🚀 Boom! Your Strategic Brief has been compiled!\n\nCheck the Final Output Window on your right. You can now use the Copy or Download PDF buttons to weaponize this logic for your corporate battle tomorrow. No wasted meetings, total ownership.'); } function updateEnergy(percentage, text) { if (energyBar) energyBar.style.width = percentage + '%'; if (energyText) energyText.innerText = text; } // Event bindings with safety checks if (safeElementCheck()) { if (sendBtn) { sendBtn.addEventListener('click', function() { var text = userInput.value.trim(); if (!text) return; appendUserMessage(text); userInput.value = ''; setTimeout(function() { handleDialogue(text); }, 600); }); } if (userInput) { userInput.addEventListener('keypress', function(e) { if (e.key === 'Enter') { e.preventDefault(); if (sendBtn) sendBtn.click(); } }); } var copyBtn = document.getElementById('copy-btn'); if (copyBtn) { copyBtn.addEventListener('click', function() { if (step < 4) { alert('Brainstorm some more before copying output!'); return; } if (!outputContent) return; navigator.clipboard.writeText(outputContent.innerText).then(function() { alert('Strategic Brief copied to clipboard! Ready to paste into PowerPoint or Word.'); }, function() { alert('Copy failed. Please select and copy manually.'); }); }); } var downloadBtn = document.getElementById('download-btn'); if (downloadBtn) { downloadBtn.addEventListener('click', function() { if (step < 4) { alert('Nothing to download yet. Finish the session first!'); return; } alert('Downloading Strategic_Brief_Report.pdf...\n\n(PoC Feature Mockup: Stamped Disclaimer included at page footer.)'); }); } window.onload = function() { appendAiMessage('Hello! I am your AI Research Partner. I am here to help filter your thoughts, handle friction, and prepare your logic for high-stakes meetings.\n\nBefore we begin, how should I address you? Please type your name below.'); }; } else { console.warn('Initialization aborted: required DOM elements missing.'); }
